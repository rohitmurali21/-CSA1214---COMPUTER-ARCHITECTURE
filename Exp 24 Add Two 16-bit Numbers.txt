.MODEL SMALL
.STACK 100H

.DATA
NUM1 DW 1234H
NUM2 DW 4321H
RESULT DW ?

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

    MOV AX, NUM1      ; Load first number into AX
    ADD AX, NUM2      ; Add second number
    MOV RESULT, AX    ; Store result

    MOV AH, 4CH
    INT 21H
MAIN ENDP
END MAIN
